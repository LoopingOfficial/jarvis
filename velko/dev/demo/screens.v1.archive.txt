export class VelkoScreenManager {
 constructor(T,meshes,bus){this.T=T;this.progress=0;this.task='En attente de votre prochaine idée';this.lines=['// VELKO · espace de travail local','// Prêt à recevoir une mission'];this.logs=['Studio initialisé','Événements en attente'];this.event='IDLE';this.tick=0;this.canvases=meshes.map((mesh,i)=>{const c=document.createElement('canvas');c.width=960;c.height=600;const tex=new T.CanvasTexture(c);tex.colorSpace=T.SRGBColorSpace;mesh.material=new T.MeshBasicMaterial({map:tex});this.textures??=[];this.textures.push(tex);const copy=c.cloneNode();document.querySelector('#screen-copies').append(copy);return c;});this.copies=[...document.querySelectorAll('#screen-copies canvas')];bus.on('*',e=>this.handle(e));this.draw(0);}
 handle(e){if(e.task)this.task=e.task;if(e.progress!==undefined)this.progress=e.progress;this.event=e.type;if(e.type==='TASK_STARTED'){this.lines=['// Mission : '+this.task,'// Simulation — aucune exécution système','const mission = {','  status: "initialisation",','  steps: []','};'];this.logs=['Nouvelle mission reçue'];}if(e.type==='FILE_OPENED')this.logs.push('Ouverture : mission.js');if(e.type==='CODE_WRITING')this.lines.push('','async function prepareMission(input) {','  const plan = await analyze(input);','  return plan.steps.map(validate);','}');if(e.type==='TERMINAL_COMMAND'){this.logs.push('$ '+e.command,'→ Vérification du scénario…');this.lines.push('','// Vérification du scénario de démonstration');}if(e.type==='ANALYSIS_RUNNING')this.logs.push('Analyse simulée des étapes…');if(e.type==='PROGRESS')this.lines.push('','mission.status = "ready";','// Plan prêt à être présenté');if(e.type==='TASK_COMPLETED')this.logs.push('✓ Démonstration terminée');}
 draw(time){this.canvases.forEach((c,i)=>{const x=c.getContext('2d'),w=c.width,h=c.height;x.fillStyle='#08121c';x.fillRect(0,0,w,h);x.fillStyle='#112334';x.fillRect(0,0,w,54);x.fillStyle='#7cc9f9';x.font='14px monospace';x.fillText('VELKO  /  '+['CODE & FICHIERS','MISSION EN COURS','SYSTÈME & ANALYTICS'][i],25,33);x.fillStyle='#51cda5';x.beginPath();x.arc(923,27,4,0,7);x.fill();
 if(i===0){x.fillStyle='#0b1926';x.fillRect(0,55,175,410);x.font='14px monospace';['EXPLORATEUR','⌄ mission','  mission.js','  journal.log','  resultat.md'].forEach((s,j)=>{x.fillStyle=j===2?'#a9dcff':'#607c90';x.fillText(s,15,95+j*32);});this.lines.forEach((s,j)=>{x.fillStyle='#344e63';x.font='13px monospace';x.fillText(String(j+1).padStart(2,'0'),189,88+j*23);x.fillStyle=s.startsWith('//')?'#63899d':s.includes('const')||s.includes('function')?'#8ac8ff':'#c1d6e1';x.fillText(s.slice(0,76),222,88+j*23);});x.fillStyle='#040b12';x.fillRect(0,462,w,138);x.font='14px monospace';this.logs.slice(-4).forEach((s,j)=>{x.fillStyle=j===3?'#63d7af':'#7e9fb4';x.fillText(s,22,490+j*26);});}
 if(i===1){x.font='13px monospace';x.fillStyle='#5da7d5';x.fillText('SIMULATEUR DE MISSION / LOCAL',35,99);x.fillStyle='#e3f0f7';x.font='25px sans-serif';const words=this.task.match(/.{1,48}(\s|$)|.{1,48}/g)||[];words.slice(0,2).forEach((s,j)=>x.fillText(s,35,145+j*32));x.font='55px sans-serif';x.fillText(this.progress+'%',35,253);x.fillStyle='#183247';x.fillRect(35,278,890,7);x.fillStyle='#3bafff';x.fillRect(35,278,890*this.progress/100,7);const steps=['Comprendre la demande','Préparer le plan','Écrire le scénario','Vérifier les étapes','Présenter le résultat'];steps.forEach((s,j)=>{let done=this.progress>j*20+10;x.fillStyle=done?'#65d5ad':'#425d72';x.font='19px sans-serif';x.fillText(done?'✓':'○',40,338+j*42);x.fillStyle=done?'#c3dbe9':'#6a8497';x.fillText(s,78,338+j*42);x.fillStyle='#223c50';x.fillRect(580,326+j*42,310,4);if(done){x.fillStyle='#339ce6';x.fillRect(580,326+j*42,310,4);}});}
 if(i===2){x.fillStyle='#789eb9';x.font='14px monospace';x.fillText('TÉLÉMÉTRIE DE DÉMONSTRATION',30,95);const labels=['PLANIFICATION','RÉDACTION','VALIDATION'];labels.forEach((s,j)=>{let cx=160+j*315;x.strokeStyle='#1c3547';x.lineWidth=9;x.beginPath();x.arc(cx,205,64,0,7);x.stroke();x.strokeStyle=['#3c9be9','#53c9c1','#acbbed'][j];x.beginPath();x.arc(cx,205,64,-Math.PI/2,-Math.PI/2+(this.progress/100)*Math.PI*2);x.stroke();x.fillStyle='#d2e6f4';x.font='28px sans-serif';x.fillText(this.progress+'%',cx-30,215);x.font='12px monospace';x.fillStyle='#789eb9';x.fillText(s,cx-55,304);});x.strokeStyle='#1c3140';x.lineWidth=1;for(let y=360;y<=520;y+=40){x.beginPath();x.moveTo(35,y);x.lineTo(925,y);x.stroke();}for(let j=0;j<35;j++){let v=30+(Math.sin(j*.72+time*.3)+1)*50+this.progress*.4;x.fillStyle=j%4===0?'#48c8c5':'#2679b2';x.fillRect(40+j*25,525-v,14,v);}x.fillStyle='#537991';x.font='13px monospace';x.fillText('FLUX ACTIF · '+this.event,35,574);}
 this.textures[i].needsUpdate=true;const cx=this.copies[i].getContext('2d');cx.drawImage(c,0,0);});}
 update(dt,time){this.tick+=dt;if(this.tick>.18){this.tick=0;this.draw(time);}}
}
