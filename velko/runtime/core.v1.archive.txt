export class VelkoEventBus extends EventTarget {
 emit(type,detail={}) {this.dispatchEvent(new CustomEvent(type,{detail}));this.dispatchEvent(new CustomEvent('*',{detail:{type,...detail}}));}
 on(type,fn){this.addEventListener(type,e=>fn(e.detail));}
}
export const STATES=['IDLE','LISTENING','THINKING','SPEAKING','MOVING_TO_WORKSTATION','SITTING','WORKING','SUCCESS','ERROR','RETURNING'];
export class VelkoStateMachine {
 constructor(bus){this.bus=bus;this.state='IDLE';this.history=[];}
 set(state){if(!STATES.includes(state))throw Error('Unknown state '+state);if(state===this.state)return;this.state=state;this.history.push({state,time:performance.now()});this.bus.emit('state.changed',{state});}
}
export class VelkoCameraDirector {
 constructor(T,camera){this.T=T;this.camera=camera;this.focus=new T.Vector3(0,1.28,2.3);this.position=new T.Vector3(0,1.47,4.4);camera.position.copy(this.position);this.mode='conversation';}
 set(mode){this.mode=mode;const views={conversation:[[0,1.37,4.15],[0,1.03,2.3]],work:[[3.5,2.05,-.1],[1.35,1.06,-2.45]],room:[[6.1,3.5,6.8],[0,.95,-.6]],transition:[[3.8,2.4,4.8],[.65,.95,.25]]};const v=views[mode]||views.conversation;this.position.set(...v[0]);this.target=new this.T.Vector3(...v[1]);}
 update(dt){this.camera.position.lerp(this.position,1-Math.exp(-dt*1.9));if(this.target)this.focus.lerp(this.target,1-Math.exp(-dt*2));this.camera.lookAt(this.focus);}
}
export class VelkoSceneDirector {
 constructor(bus,machine,avatar,camera,speak){Object.assign(this,{bus,machine,avatar,camera,speak});this.active=false;this.time=0;this.next=0;this.marks=[];this.context={seated:true};avatar.root.position.set(0,0,2.3);this.mode='conversation';}
 schedule(t,fn){this.marks.push({t,fn});}
 start(task,automatic=false){if(this.active)return false;this.active=true;this.time=0;this.next=0;this.marks=[];this.task=task;this.bus.emit('TASK_STARTED',{task,progress:0});this.machine.set('IDLE');this.camera.set('conversation');
 this.schedule(automatic?3:0,()=>this.machine.set('LISTENING'));
 this.schedule(automatic?5:1.5,()=>this.machine.set('THINKING'));
 this.schedule(automatic?7:3,()=>{this.machine.set('SPEAKING');this.speak('Bien reçu. Je vais préparer une démonstration de cette mission à mon poste.');});
 const shift=automatic?0:-4;
 this.schedule(9+shift,()=>{this.context.look=1;});
 this.schedule(10+shift,()=>{this.machine.set('MOVING_TO_WORKSTATION');this.context.seated=false;this.camera.set('transition');this.moveStart=this.time;});
 this.schedule(17+shift,()=>{this.machine.set('SITTING');this.context.walking=false;this.context.seated=true;this.camera.set('work');});
 this.schedule(19+shift,()=>{this.machine.set('WORKING');this.context.typing=true;this.bus.emit('FILE_OPENED',{file:'mission.js',progress:12});});
 this.schedule(21+shift,()=>{this.context.look=-1;this.bus.emit('CODE_WRITING',{progress:29,task});});
 this.schedule(24+shift,()=>{this.context.look=0;this.bus.emit('TERMINAL_COMMAND',{command:'velko demo --validate',progress:46});});
 this.schedule(27+shift,()=>{this.context.look=1;this.context.typing=false;this.context.mouse=true;this.bus.emit('ANALYSIS_RUNNING',{progress:67});});
 this.schedule(30+shift,()=>{this.context.look=0;this.context.typing=true;this.context.mouse=false;this.bus.emit('PROGRESS',{progress:88});});
 this.schedule(33+shift,()=>{this.machine.set('SUCCESS');this.context.typing=false;this.bus.emit('TASK_COMPLETED',{progress:100,result:'Démonstration terminée. Plan, étapes et journal prêts à consulter.'});});
 this.schedule(35+shift,()=>{this.machine.set('RETURNING');this.context.seated=false;this.returnStart=this.time;this.camera.set('transition');});
 this.schedule(42+shift,()=>{this.context.walking=false;this.context.seated=true;this.context.look=0;this.camera.set('conversation');this.machine.set('SPEAKING');this.speak('La démonstration est terminée. Vous pouvez consulter le journal et les trois écrans. Je suis prêt pour la suite.');});
 this.schedule(46+shift,()=>{this.machine.set('IDLE');this.active=false;this.bus.emit('demo.finished');});return true;}
 update(dt){if(this.active){this.time+=dt;while(this.next<this.marks.length&&this.time>=this.marks[this.next].t)this.marks[this.next++].fn();}
 const a=this.avatar.root,s=this.machine.state;this.context.speaking=s==='SPEAKING';
 if(s==='MOVING_TO_WORKSTATION'||s==='RETURNING'){
 const returning=s==='RETURNING',u=Math.min(1,(this.time-(returning?this.returnStart:this.moveStart))/7);const walk=Math.max(0,Math.min(1,(u-.2)/.68));this.context.walking=walk>0&&walk<1;
 const p=returning?1-walk:walk;const bend=.8;let dx,dz;if(p<bend){a.position.set(.62*p/bend,0,2.3-3.9*p/bend);dx=.62;dz=-3.9;}else{a.position.set(.62+.78*(p-bend)/(1-bend),0,-1.6);dx=.78;dz=0;}
 const desired=Math.atan2(returning?-dx:dx,returning?-dz:dz);const target=walk>=1?(returning?0:Math.PI):desired;
 let diff=((target-a.rotation.y+Math.PI)%(2*Math.PI)+2*Math.PI)%(2*Math.PI)-Math.PI;a.rotation.y+=diff*(1-Math.exp(-dt*5));
 }this.avatar.update(dt,performance.now()/1000,s.toLowerCase(),this.context);}
}
